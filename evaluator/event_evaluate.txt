<?php
          $count = 0;
          while ($q = mysqli_fetch_assoc($questions)):
            $count++;
            $opts = $q['options'] ? json_decode($q['options'], true) : [];
            ?>
            <div class="mb-4">
              <label class="form-label fw-bold">Q<?php echo $count; ?>:
                <?php echo htmlspecialchars($q['question_text']); ?></label>

              <?php if ($q['question_type'] == 'rating'): ?>
                <div>
                  <?php for ($i = 1; $i <= 5; $i++): ?>
                    <label class="me-2">
                      <input type="radio" name="answers[<?php echo $q['question_id']; ?>]" value="<?php echo $i; ?>" required>
                      <?php echo $i; ?>
                    </label>
                  <?php endfor; ?>
                </div>

              <?php elseif ($q['question_type'] == 'multiple'): ?>
                <select name="answers[<?php echo $q['question_id']; ?>]" class="form-select" required>
                  <option value="">Select an option</option>
                  <?php foreach ($opts as $opt): ?>
                    <option value="<?php echo htmlspecialchars($opt); ?>"><?php echo htmlspecialchars($opt); ?></option>
                  <?php endforeach; ?>
                </select>

              <?php else: ?>
                <textarea name="answers[<?php echo $q['question_id']; ?>]" class="form-control" rows="2"
                  placeholder="Enter your response" required></textarea>
              <?php endif; ?>
            </div>
          <?php endwhile; ?>
