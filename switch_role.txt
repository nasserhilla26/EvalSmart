<?php
session_start();
// file_put_contents('session_debug.log', print_r($_SESSION, true)); //debug purposes checking session roles
header('Content-Type: application/json');

// Disable warnings/notices from breaking JSON
error_reporting(E_ERROR | E_PARSE);
ob_clean();

$response = ['success' => false, 'message' => 'Unknown error'];

try {
    // Check session
    if (!isset($_SESSION['user_id'])) {
        throw new Exception('Session expired. Please log in again.');
    }

    // Check roles exist
    if (!isset($_SESSION['roles']) || !is_array($_SESSION['roles'])) {
        throw new Exception('Roles not found in session.');
    }

    // Validate input
    if (!isset($_POST['role_id']) || !in_array((int)$_POST['role_id'], $_SESSION['roles'])) {
        throw new Exception('Invalid or unauthorized role.');
    }

    $selected_role = (int)$_POST['role_id'];

    // Block organizer if not approved
    if ($selected_role === 2) {

        include '../includes/db_connect.php';

        $user_id = $_SESSION['user_id'];

        $check = mysqli_query($conn, "
            SELECT role_status FROM users WHERE user_id = $user_id
        ");

        $user = mysqli_fetch_assoc($check);

        if ($user['role_status'] !== 'Approved') {
            throw new Exception('Your organizer account is still pending approval.');
        }
    }


    // Refresh roles from DB (IMPORTANT FIX)
    include '../includes/db_connect.php';

    $user_id = $_SESSION['user_id'];

    $new_roles = [];

    $role_query = mysqli_query($conn, "
        SELECT role_id FROM user_roles WHERE user_id = $user_id
    ");

    $user_check = mysqli_query($conn, "
        SELECT role_status FROM users WHERE user_id = $user_id
    ");
    $user = mysqli_fetch_assoc($user_check);

    while ($r = mysqli_fetch_assoc($role_query)) {

        // Hide organizer if not approved
        if ($r['role_id'] == 2 && $user['role_status'] != 'Approved') {
            continue;
        }

        $new_roles[] = (int)$r['role_id'];
    }

    // Update session roles
    $_SESSION['roles'] = $new_roles;

    // Assign active role
    $_SESSION['active_role'] = (int)$_POST['role_id'];


    // Redirect mapping
    switch ($_SESSION['active_role']) {
        case 1:
            $redirect = '/evalsmart/admin/dashboard.php';
            break;
        case 2:
            $redirect = '/evalsmart/organizer/dashboard.php';
            break;
        case 3:
            $redirect = '/evalsmart/evaluator/dashboard.php';
            break;
        default:
            throw new Exception('Invalid role ID.');
    }

    $response = [
        'success' => true,
        'redirect' => $redirect,
        'active_role' => $_SESSION['active_role']
    ];

} catch (Exception $e) {
    $response['message'] = $e->getMessage();
}

// Return JSON cleanly
echo json_encode($response);
exit;
?>
